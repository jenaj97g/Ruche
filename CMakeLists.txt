cmake_minimum_required(VERSION 3.20)
project(ruche VERSION 0.1.9 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
# The Windows cross build uses the host rcc, whose zstd support the target Qt may lack.
set(CMAKE_AUTORCC_OPTIONS --no-zstd)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# D-Bus (MPRIS, desktop notifications, single instance) everywhere but Windows.
if(WIN32)
  set(RUCHE_DBUS_DEFAULT OFF)
else()
  set(RUCHE_DBUS_DEFAULT ON)
endif()
option(RUCHE_DBUS "Use D-Bus for MPRIS, notifications and single-instance handover" ${RUCHE_DBUS_DEFAULT})

set(QT_COMPONENTS Core Gui Widgets Network)
if(RUCHE_DBUS)
  list(APPEND QT_COMPONENTS DBus)
endif()
find_package(Qt6 REQUIRED COMPONENTS ${QT_COMPONENTS})
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
pkg_check_modules(TAGLIB REQUIRED IMPORTED_TARGET taglib)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)

set(RUCHE_UPDATE_REPO "jenaj97g/Ruche" CACHE STRING "GitHub repository (owner/name) whose releases updates come from")
set(RUCHE_DISCORD_CLIENT_ID "" CACHE STRING "Discord application ID for Rich Presence (discord.com/developers)")

# Lua 5.4 for plugins (vendored, MIT)
file(GLOB LUA_SOURCES ${CMAKE_SOURCE_DIR}/src/3rdparty/lua/*.c)
add_library(ruche_lua STATIC ${LUA_SOURCES})
set_target_properties(ruche_lua PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
if(NOT WIN32)
  target_compile_definitions(ruche_lua PRIVATE LUA_USE_POSIX)
endif()
if(NOT WIN32)
  target_link_libraries(ruche_lua PRIVATE m)
endif()

add_library(ruche_meta STATIC src/MetaParse.cpp)
target_link_libraries(ruche_meta PUBLIC Qt6::Core)

set(RUCHE_SOURCES
  src/main.cpp
  src/Player.cpp
  src/HlsProxy.cpp
  src/StreamResolver.cpp
  src/Library.cpp
  src/Art.cpp
  src/PlaybackController.cpp
  src/Models.cpp
  src/Icons.cpp
  src/Theme.cpp
  src/ThemeFiles.cpp
  src/CrashLog.cpp
  src/CoverLookup.cpp
  src/Stations.cpp
  src/Logo.cpp
  src/Mpris.cpp
  src/DesktopEntry.cpp
  src/Widgets.cpp
  src/MainWindow.cpp
  src/Notifier.cpp
  src/NowPlaying.cpp
  src/DiscordPresence.cpp
  src/Plugins.cpp
  src/Updater.cpp
  src/I18n.cpp
  i18n/i18n.qrc
)
if(WIN32)
  # packaging/windows/ruche.ico is generated by packaging/windows/build-windows.sh
  configure_file(packaging/windows/ruche.rc.in ${CMAKE_BINARY_DIR}/ruche.rc @ONLY)
  list(APPEND RUCHE_SOURCES ${CMAKE_BINARY_DIR}/ruche.rc)
endif()

add_executable(ruche ${RUCHE_SOURCES})
target_compile_definitions(ruche PRIVATE RUCHE_VERSION="${PROJECT_VERSION}" QT_NO_KEYWORDS
  RUCHE_DISCORD_CLIENT_ID="${RUCHE_DISCORD_CLIENT_ID}" RUCHE_UPDATE_REPO="${RUCHE_UPDATE_REPO}")
target_link_libraries(ruche PRIVATE ruche_meta ruche_lua Qt6::Widgets Qt6::Network
  PkgConfig::MPV PkgConfig::TAGLIB PkgConfig::SQLITE)
# Theme import reads .zip files with Qt's own (private) zip reader: in QtGui up to
# Qt 6.6, in QtCore from 6.7.
foreach(_priv CorePrivate GuiPrivate)
  if(TARGET Qt6::${_priv})
    target_link_libraries(ruche PRIVATE Qt6::${_priv})
  endif()
endforeach()
if(RUCHE_DBUS)
  target_compile_definitions(ruche PRIVATE RUCHE_DBUS)
  target_link_libraries(ruche PRIVATE Qt6::DBus)
endif()
if(WIN32)
  set_target_properties(ruche PROPERTIES WIN32_EXECUTABLE ON OUTPUT_NAME Ruche)
  target_compile_definitions(ruche PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MINGW)
    target_link_options(ruche PRIVATE -Wl,-O1)
  endif()
else()
  target_link_options(ruche PRIVATE -Wl,--as-needed -Wl,-O1)
endif()
install(TARGETS ruche RUNTIME DESTINATION bin)
# example plugins, listed (disabled) in Settings → Plugins
if(WIN32)
  install(DIRECTORY plugins/ DESTINATION bin/plugins)
  install(DIRECTORY themes/ DESTINATION bin/themes)
else()
  install(DIRECTORY plugins/ DESTINATION share/ruche/plugins)
  install(DIRECTORY themes/ DESTINATION share/ruche/themes)
  install(FILES PLUGINS.md THEMES.md DESTINATION share/ruche)
endif()

add_executable(test_meta tests/test_meta.cpp)
target_link_libraries(test_meta PRIVATE ruche_meta)
